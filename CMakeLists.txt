cmake_minimum_required(VERSION 3.25)
project(OpenBK2Launcher VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(OBK_FIXTURE_BUILD "Build a visibly identified development launcher" OFF)
option(OBK_BUILD_GUI "Build the native SDL3/ImGui application" ON)
set(OBK_MANIFEST_URL "" CACHE STRING "Production HTTPS signed envelope endpoint")
set(OBK_PUBLIC_KEY_HEX "" CACHE STRING "Production Ed25519 public key, 64 hex characters")
set(OBK_KEY_ID "" CACHE STRING "Production signing key identifier")
if(NOT OBK_FIXTURE_BUILD)
  string(LENGTH "${OBK_PUBLIC_KEY_HEX}" key_length)
  if(NOT OBK_MANIFEST_URL MATCHES "^https://[^/]+/" OR
     NOT key_length EQUAL 64 OR NOT OBK_PUBLIC_KEY_HEX MATCHES "^[0-9a-fA-F]+$" OR
     NOT OBK_KEY_ID MATCHES "^[A-Za-z0-9_-]+$")
    message(FATAL_ERROR "Production builds require OBK_MANIFEST_URL (HTTPS), OBK_KEY_ID, and OBK_PUBLIC_KEY_HEX. Use the fixture preset for development.")
  endif()
endif()
if(OBK_MANIFEST_URL MATCHES "[\"\\\\\r\n]")
  message(FATAL_ERROR "Manifest URL contains invalid configuration characters")
endif()
configure_file(cmake/build_config.hpp.in generated/build_config.hpp @ONLY)

include(CTest)
include(cmake/Dependencies.cmake)
obk_security_dependencies()
obk_transfer_dependencies()
add_library(launcher_core STATIC src/window_layout.cpp src/sha256.cpp src/signature.cpp src/manifest.cpp
  src/safe_files.cpp src/http_downloader.cpp src/zip_package_reader.cpp)
target_sources(launcher_core PRIVATE src/local_state.cpp src/update_planner.cpp src/transaction.cpp src/installer.cpp)
target_sources(launcher_core PRIVATE src/platform_services.cpp src/settings.cpp src/controller.cpp)
target_include_directories(launcher_core PRIVATE ${CMAKE_BINARY_DIR}/generated)
if(WIN32)
  target_link_libraries(launcher_core PRIVATE ole32 shell32 uuid)
endif()
target_link_libraries(launcher_core PRIVATE yyjson monocypher picosha miniz CURL::libcurl)
target_compile_definitions(launcher_core PRIVATE OBK_FIXTURE_BUILD=$<BOOL:${OBK_FIXTURE_BUILD}>)
target_include_directories(launcher_core PUBLIC include)
target_compile_features(launcher_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(launcher_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(launcher_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(OBK_BUILD_GUI)
  obk_gui_dependencies()
  file(READ "${OBK_FONT_SOURCE}" font_hex HEX)
  string(REGEX REPLACE "(..)" "0x\\1," font_bytes "${font_hex}")
  file(WRITE "${CMAKE_BINARY_DIR}/generated/font.hpp" "#pragma once\ninline unsigned char embedded_font[]={${font_bytes}};\n")
  add_executable(OpenBK2Launcher src/main.cpp src/application.cpp)
  target_include_directories(OpenBK2Launcher PRIVATE ${CMAKE_BINARY_DIR}/generated)
  target_link_libraries(OpenBK2Launcher PRIVATE launcher_core imgui SDL3::SDL3-static)
  target_compile_definitions(OpenBK2Launcher PRIVATE
    OBK_VERSION="${PROJECT_VERSION}" OBK_FIXTURE_BUILD=$<BOOL:${OBK_FIXTURE_BUILD}>)
  if(WIN32)
    enable_language(RC)
    configure_file(assets/windows.rc.in generated/windows.rc @ONLY)
    target_sources(OpenBK2Launcher PRIVATE ${CMAKE_BINARY_DIR}/generated/windows.rc)
    set_target_properties(OpenBK2Launcher PROPERTIES WIN32_EXECUTABLE TRUE)
    set(OBK_PLATFORM Windows-x64)
    target_compile_options(OpenBK2Launcher PRIVATE /utf-8 /W4 /constexpr:steps5000000)
    target_link_options(OpenBK2Launcher PRIVATE $<$<NOT:$<CONFIG:Debug>>:/OPT:REF> $<$<NOT:$<CONFIG:Debug>>:/OPT:ICF>)
  else()
    set(OBK_PLATFORM Linux-x64)
    target_compile_options(OpenBK2Launcher PRIVATE -Wall -Wextra -Wpedantic $<$<NOT:$<CONFIG:Debug>>:-Os>)
    target_link_options(OpenBK2Launcher PRIVATE $<$<NOT:$<CONFIG:Debug>>:-s> $<$<NOT:$<CONFIG:Debug>>:-Wl,--gc-sections>)
  endif()
  set_target_properties(OpenBK2Launcher PROPERTIES OUTPUT_NAME "OpenBK2Launcher-${OBK_PLATFORM}")
  add_custom_command(TARGET OpenBK2Launcher POST_BUILD
    COMMAND ${CMAKE_COMMAND} -DBINARY=$<TARGET_FILE:OpenBK2Launcher> -P ${CMAKE_SOURCE_DIR}/cmake/CheckSize.cmake)
  install(TARGETS OpenBK2Launcher RUNTIME DESTINATION .)
  if(BUILD_TESTING)
    add_test(NAME renderer_smoke COMMAND OpenBK2Launcher --smoke-test)
    set_tests_properties(renderer_smoke PROPERTIES ENVIRONMENT "SDL_VIDEO_DRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 20)
    add_test(NAME binary_size COMMAND ${CMAKE_COMMAND} -DBINARY=$<TARGET_FILE:OpenBK2Launcher> -P ${CMAKE_SOURCE_DIR}/cmake/CheckSize.cmake)
  endif()
endif()
if(BUILD_TESTING)
  find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
  add_executable(core_tests tests/core/window_layout_test.cpp)
  target_link_libraries(core_tests PRIVATE launcher_core)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(manifest_check tests/core/manifest_check.cpp)
  target_link_libraries(manifest_check PRIVATE launcher_core)
  add_executable(security_tests tests/core/security_test.cpp)
  target_link_libraries(security_tests PRIVATE launcher_core)
  add_test(NAME security_primitives COMMAND security_tests)
  add_test(NAME publisher_and_manifest COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tests/integration/publisher_manifest.py $<TARGET_FILE:manifest_check>)
  set_tests_properties(publisher_and_manifest PROPERTIES TIMEOUT 120)
  add_test(NAME github_publication COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/github_publication.py)
  set_tests_properties(github_publication PROPERTIES TIMEOUT 120)
  add_executable(transfer_check tests/core/transfer_check.cpp)
  target_link_libraries(transfer_check PRIVATE launcher_core)
  add_test(NAME transfer_and_staging COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tests/integration/transfer_staging.py $<TARGET_FILE:transfer_check>)
  set_tests_properties(transfer_and_staging PROPERTIES TIMEOUT 120)
  add_executable(installer_check tests/core/installer_check.cpp)
  target_link_libraries(installer_check PRIVATE launcher_core)
  add_test(NAME installer_recovery COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tests/integration/installer_recovery.py $<TARGET_FILE:installer_check>)
  set_tests_properties(installer_recovery PROPERTIES TIMEOUT 180)
  add_executable(fixture_game tests/core/fixture_game.cpp)
  target_link_libraries(fixture_game PRIVATE launcher_core)
  add_executable(lifecycle_check tests/core/lifecycle_check.cpp)
  target_link_libraries(lifecycle_check PRIVATE launcher_core)
  if(OBK_BUILD_GUI AND OBK_FIXTURE_BUILD)
    add_test(NAME launcher_lifecycle COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_SOURCE_DIR}/tests/integration/launcher_lifecycle.py $<TARGET_FILE:lifecycle_check> $<TARGET_FILE:fixture_game> $<TARGET_FILE:OpenBK2Launcher>)
    set_tests_properties(launcher_lifecycle PROPERTIES TIMEOUT 120)
  endif()
endif()
